import { Tabs, TabItem } from '@astrojs/starlight/components';

<Tabs syncKey="os">
	<TabItem label="Windows" icon="seti:windows">
		1. Open **Settings → Bluetooth & devices → Printers & scanners**.
		2. Select **Add device** and choose the printer by its name, for example **[FLOOR2-PRINT]**.
	</TabItem>
	<TabItem label="macOS" icon="apple">
		1. Open **System Settings → Printers & Scanners**.
		2. Select **Add Printer, Scanner or Fax** and choose the printer.
	</TabItem>
</Tabs>

Printer names are on a label on each printer. For jams or empty toner, see the instructions on the printer or [report it](/support/get-help/).
